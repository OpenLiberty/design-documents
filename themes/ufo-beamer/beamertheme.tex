% ── Changebars ────────────────────────────────────────────────────────────────
% Uses a TikZ remember picture/overlay approach rather than the changebar package
% so the bar position is always fixed in the slide margin, independent of list
% indent depth.
%
% \cbstart{} records the current page position into a named TikZ coordinate.
% \cbend{} records the end position and draws a red vertical rule between them
% in an overlay, anchored 0.85cm from the slide's left paper edge.
%
% ulem provides \sout{} for struck-through deleted text.
%
% Usage in markdown:
%
%   Block-level (whole bullet or paragraph):
%     ::: changed
%     - Modified bullet
%     :::
%
%   Inline addition:
%     [new wording]{.added}
%
%   Inline deletion (struck through):
%     [old wording]{.deleted}
%
\usepackage[normalem]{ulem}   % \sout — normalem keeps \emph as italic
\usepackage{zref-savepos}     % \zsavepos — records absolute y position on page

% \ufoAdded{text} — inline highlight for added/changed text.
% Uses a very light magenta tint (ufochanged!15) with tight padding so the
% highlight is visible without disrupting line spacing.
% \fboxsep controls the padding; saving/restoring avoids side-effects.
\newcommand{\ufoAdded}[1]{%
  {\setlength{\fboxsep}{1.5pt}\colorbox{ufochanged!15}{#1}}%
}

% CVD-safe changebar colour: IBM Magenta-50 #EE538B
% Distinguishable from navy/teal/blue family by both hue and luminance
% across all CVD types (protanopia, deuteranopia, tritanopia).
\definecolor{ufochanged}{RGB}{238,83,139}

% Counter to give each changebar a unique label pair
\newcounter{cbcount}

\newcommand{\cbstart}{%
  \stepcounter{cbcount}%
  \zsavepos{cb-start-\thecbcount}%
}

\newcommand{\cbend}{%
  \zsavepos{cb-end-\thecbcount}%
  \edef\cbcurrent{\thecbcount}%
  % Compute y coordinates in pt: zref stores sp (1pt = 65536sp)
  \edef\cbystart{\the\dimexpr\zposy{cb-start-\cbcurrent} sp\relax}%
  \edef\cbyend{\the\dimexpr\zposy{cb-end-\cbcurrent} sp\relax}%
  \begin{tikzpicture}[remember picture, overlay]
    % Draw a red vertical rule at x = 0.85cm from the left paper edge,
    % spanning from the \cbend position (bottom) to the \cbstart position (top).
    \draw[ufochanged, line width=2.5pt, line cap=round]
      ([xshift=0.85cm, yshift=\cbyend] current page.south west)
      --
      ([xshift=0.85cm, yshift=\cbystart] current page.south west);
  \end{tikzpicture}%
}

% ── Title page ────────────────────────────────────────────────────────────────
% Layout derived from PPTX EMU coordinates + pixel calibration against title-bg.png
% (3584×2016 px, 300 dpi → 254mm×142.875mm when scaled to paperwidth):
%   - title-bg.png: full-bleed dark navy + yellow-green UFO searchlight
%   - Open Liberty logo: top-left, width=6.291cm, offset (0.647cm, −0.826cm)
%   - Title text: xshift=9% from left, yshift=−48% from top (inside beam at left edge)
%   - Author block: xshift=9% from left, ~58% from top (raisebox −0.26 from text-area top)
%   - URL footnote: bottom-left corner, 15pt dark navy
%
% NOTE: TikZ remember picture/overlay requires TWO xelatex passes to resolve
% current page anchors correctly (see Makefile).  With a single pass the
% background images render offset.
%
% \href inside TikZ nodes is dropped by xdvipdfmx; the author block therefore
% uses \rlap/\raisebox in normal TeX flow to preserve live hyperlinks.
% The raisebox origin is the top-left of the frame text area (≈30% from page top
% on the title frame); −0.26\paperheight drops it to ≈58% from page top.
\definecolor{ufolink}{RGB}{0,102,204}   % blue — distinguishable from the teal body text

\setbeamertemplate{title page}{%

  \begin{tikzpicture}[remember picture,overlay]
    \node[at=(current page.center)] {%
      \includegraphics[width=\paperwidth,height=\paperheight]{title-bg.png}%
    };
    \node[anchor=north west, xshift=0.647cm, yshift=-0.826cm]
      at (current page.north west) {%
      \includegraphics[width=6.291cm]{openliberty-logo.png}%
    };
    \node[anchor=north west,
          xshift=0.09\paperwidth,
          yshift=-0.48\paperheight] at (current page.north west) {%
      \begin{minipage}{0.62\paperwidth}
        \raggedright
        \usebeamerfont{title}\titlefont\color{ufonavydark}%
        \inserttitle\ufodraftbadge%
      \end{minipage}%
    };
    % ── Draft instruction (shown only when dirty) ──────────────────────────
    \ufodraftwarning%
    \node[anchor=south west, xshift=0.3cm, yshift=0.25cm]
      at (current page.south west) {%
      \fontsize{15}{17}\selectfont\color{ufonavydark}%
      Upcoming Feature Overview (UFO) Template ---
      https://github.com/OpenLiberty/design-documents/tree/ufo-beamer%
    };
  \end{tikzpicture}%
  % ── Layer 2: author block as normal TeX flow (outside TikZ for live \href) ───
  % \href inside TikZ nodes is dropped by xdvipdfmx; author must stay outside TikZ.
  % After \end{tikzpicture} TeX is in horizontal mode with a zero-size box at the
  % top-left of the frame text area (≈30% from page top on the title frame).
  % \rlap keeps zero width so no space is consumed; \raisebox{-0.26\paperheight}
  % drops the block to ≈56% from page top, below the title node.
  \rlap{%
    \raisebox{-0.26\paperheight}[0pt][0pt]{%
      \hspace*{0.09\paperwidth}%
      \begin{minipage}{0.55\paperwidth}
        \raggedright
        \usebeamerfont{author}\color{ufoteal}%
        \insertauthor
      \end{minipage}%
    }%
  }%
  % ── Layer 3: optional change key (drawn only when \ufochangekey was called) ──
  \ufochangekeyifenabled%
}

% ── Section divider slide ─────────────────────────────────────────────────────
% Usage in markdown:
%
%   # Section Title {.unnumbered}
%
%   ```{=latex}
%   \sectionslide{Section Title}
%   ```
%
% IMPORTANT: Do NOT use {.plain} — the footline must remain active so the page
% number appears. \sectionslide draws the full-bleed dark background and centred
% title via TikZ (covering the default frametitle), then overrides the footline
% colour to lime for this frame only.
\newcommand{\sectionslide}[1]{%
  \setbeamercolor{footline}{fg=ufolime}%
  % Suppress the frametitle vskip for this frame so no white gap appears above
  % the full-bleed TikZ background overlay.
  \setbeamertemplate{frametitle}{}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[at=(current page.center)] {%
      \includegraphics[width=\paperwidth,height=\paperheight]{section-bg.png}%
    };
    \node[at=(current page.center)] {%
      \begin{minipage}{0.75\paperwidth}
        \centering
        \titlefont\fontsize{54}{60}\selectfont\color{white}#1
      \end{minipage}%
    };
  \end{tikzpicture}%
  \setbeamercolor{footline}{fg=ufopagenumber}%
}

% ── Instruction notes environment ────────────────────────────────────────────
% ::: instruction blocks in markdown compile to \begin{instruction}...\end{instruction}
% via pandoc's generic div handling (via theme/instruction-filter.lua).
% On the presentation slides these produce no visible output; their content is
% surfaced only in the notes/handout PDFs generated by generate-handouts.py,
% where they appear in an amber-tinted box.
%
% Implementation: collect the body into a scratch vbox that is immediately
% discarded — zero output, zero spacing, safe inside beamer frames (including
% [fragile] frames and TikZ overlay frames).
\newenvironment{instruction}{\setbox0\vbox\bgroup\hsize\textwidth}{\egroup}

% ── Deleted block environment ─────────────────────────────────────────────────
% ::: deleted divs in markdown compile to \begin{ufoDeletedBlock}...\end{ufoDeletedBlock}
% via changebar-filter.lua.
%
% Behaviour:
%   \ifufochanges (changebar/review build): content is rendered in faded magenta
%     with a margin changebar. The Lua filter (changebar-filter.lua) has already
%     wrapped every bullet's inline text in \sout{} at the Pandoc AST level.
%   otherwise (final build): content is silently discarded — same vbox trick as
%     \begin{instruction}, safe inside any beamer frame.
\newenvironment{ufoDeletedBlock}{%
  \ifufochanges
    \cbstart{}%
  \else
    \setbox0\vbox\bgroup\hsize\textwidth
  \fi
}{%
  \ifufochanges
    \cbend{}%
  \else
    \egroup
  \fi
}
