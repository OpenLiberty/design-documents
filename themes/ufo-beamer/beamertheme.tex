% ── Block-level changebar mechanism (for ::: changed / added / deleted divs) ──
%
% Uses zref-savepos to record absolute Y positions at \cbstart and \cbend.
% A LIFO stack (cbdepth counter + per-depth ID storage) lets nested pairs
% work correctly: the outer \cbend retrieves its own ID even when inner
% \cbstart calls have advanced the cbcount counter.
%
% !! Only works reliably when start and end are on DIFFERENT baselines !!
% For single-line inline spans use \cbistart / \cbiend instead (see below).
%
% Top-of-bar correction: pandoc always emits \cbstart{} as a standalone
% RawBlock, so a blank line (= \par) follows it before the content block.
% TeX deposits \parskip glue AFTER the \zsavepos call, so the saved Y is
% one \parskip above the first bullet.  \cbend corrects for this by storing
% the natural part of \parskip at \cbstart time and subtracting it when
% drawing the bar's top endpoint.
%
\newcounter{cbcount}   % ever-incrementing unique ID for each \cbstart
\newcounter{cbdepth}   % current nesting depth (0 = not inside any bar)
\newdimen\cbpsnat      % scratch dimen — natural part of \parskip at \cbstart

\newcommand{\cbstart}{%
  \stepcounter{cbcount}%
  \stepcounter{cbdepth}%
  \expandafter\xdef\csname cbid@\thecbdepth\endcsname{\thecbcount}%
  % Snapshot the natural part of \parskip so \cbend can correct the bar top.
  \cbpsnat=\parskip
  \expandafter\xdef\csname cbps@\thecbcount\endcsname{\the\cbpsnat}%
  \zsavepos{cb-start-\thecbcount}%
}

\newcommand{\cbend}{%
  \edef\cbcurrent{\csname cbid@\thecbdepth\endcsname}%
  \zsavepos{cb-end-\cbcurrent}%
  \addtocounter{cbdepth}{-1}%
  \ifufochanges%
    \edef\cbystart{\the\dimexpr\zposy{cb-start-\cbcurrent} sp\relax}%
    \edef\cbyend{\the\dimexpr\zposy{cb-end-\cbcurrent} sp\relax}%
    \edef\cbpsval{\csname cbps@\cbcurrent\endcsname}%
    \begin{tikzpicture}[remember picture, overlay]
      \draw[ufochanged, line width=2.5pt, line cap=round]
        ([xshift=0.85cm, yshift=\cbyend] current page.south west)
        --
        ([xshift=0.85cm, yshift=\dimexpr\cbystart - \cbpsval\relax]
          current page.south west);
    \end{tikzpicture}%
  \fi%
}

% ── Inline changebar mechanism (for {.added} / {.deleted} / {.changed} spans) ─
%
% Problem: \zsavepos at the start of a list-item paragraph records the Y of
% the PREVIOUS line (TeX has not yet committed the current line when in
% horizontal mode at a paragraph's start), so the bar appears one line too high.
%
% Solution: both \cbistart and \cbiend save their positions inside \vadjust,
% which fires AFTER the current line is broken and shipped — giving the correct
% baseline Y for the line containing the span.  \cbiend reads the position
% saved by \cbistart (from the previous xelatex pass) and draws the bar.
% Four xelatex passes (as run by the Makefile) are sufficient to converge.
%
% Nesting: same LIFO stack as block bars — cbdepth / cbid@ — so a
% {.changed} span wrapping {.deleted}+{.added} works without conflict.
%
\newcommand{\cbistart}{%
  \stepcounter{cbcount}%
  \stepcounter{cbdepth}%
  \expandafter\xdef\csname cbid@\thecbdepth\endcsname{\thecbcount}%
  % \leavevmode ensures TeX is in horizontal mode before \zsavepos fires,
  % so the recorded Y is this line's baseline, not the previous line's.
  % (At paragraph start, without \leavevmode, zref records the pre-paragraph
  % vertical position — one line too high.)
  \leavevmode\zsavepos{cbi-start-\thecbcount}%
}

\newcommand{\cbiend}{%
  \edef\cbicurrent{\csname cbid@\thecbdepth\endcsname}%
  \addtocounter{cbdepth}{-1}%
  \ifufochanges%
    \edef\cbiy{\the\dimexpr\zposy{cbi-start-\cbicurrent} sp\relax}%
    % \vadjust inserts material into the vertical list after the current line.
    % We draw zero-height material so it does not affect spacing.
    \vadjust{%
      \begin{tikzpicture}[remember picture, overlay]
        \draw[ufochanged, line width=2.5pt, line cap=round]
          ([xshift=0.85cm,
            yshift=\dimexpr\cbiy - \dp\strutbox\relax]
            current page.south west)
          --
          ([xshift=0.85cm,
            yshift=\dimexpr\cbiy + \ht\strutbox\relax]
            current page.south west);
      \end{tikzpicture}%
    }%
  \fi%
}
